cmake_minimum_required(VERSION 3.20)
project(cinder VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SANITIZER "" CACHE STRING "address or thread")
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(SANITIZER)
  add_compile_options(-fsanitize=${SANITIZER} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${SANITIZER})
endif()
find_package(Threads REQUIRED)
add_library(cinder_core src/store.cpp src/protocol.cpp)
target_include_directories(cinder_core PUBLIC src)
target_link_libraries(cinder_core PUBLIC Threads::Threads)
add_executable(cinder src/server.cpp)
target_link_libraries(cinder PRIVATE cinder_core)
add_executable(core_tests tests/core.cpp)
target_link_libraries(core_tests PRIVATE cinder_core)
enable_testing()
add_test(NAME core COMMAND core_tests)
